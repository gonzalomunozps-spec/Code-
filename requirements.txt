# Dependencias del Sistema de Gestion y Monitoreo de Parcelas
# Instalar con:  pip install -r requirements.txt
#
# NOTA: tkinter NO se instala con pip (viene con Python en Windows/macOS; en
# Linux es el paquete del sistema 'python3-tk'). Ver INSTALL / README.

PySide6              # interfaz Qt (panel_qt.py). LGPL, a diferencia de PyQt, que es
                     # GPL o comercial. En Linux necesita ademas las librerias del
                     # sistema libegl1, libgl1 y libxkbcommon-x11-0.
earthengine-api      # Google Earth Engine: descarga de imagenes Sentinel-2
tkintermapview       # mapa interactivo para dibujar/buscar parcelas
pillow               # imagenes (mapa de indices)
matplotlib           # graficas de evolucion y leyendas de color
requests             # peticiones HTTP (SIGPAC, descarga de tiles)
openai               # OPCIONAL: interpretacion con ChatGPT (sin clave, va por reglas)
reportlab            # OPCIONAL: informes en PDF (balance y tecnico) del modulo informe_anual.py
openpyxl             # OPCIONAL: exportacion a Excel (indices por mes + graficas) del mismo modulo
platformdirs         # OPCIONAL: ubica los datos en la carpeta estandar del sistema (si falta, ~/.gestor_parcelas)
